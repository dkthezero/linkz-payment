# Checkout Flow Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the checkout step (review → pay → poll) plus the reservations backend module it depends on, integrated with the landed seat-map UI, with Playwright E2E as TDD.

**Architecture:** New `ReservationsModule` (atomic hold + review + release) is the foundation; `PaymentsModule` is fixed to authorize against the reservation, charge `seats.price + flat fee`, persist the Stripe session id, and expose a poll endpoint; a SolidJS router adds Checkout/Success/Cancel screens wired to the seat-map's "Proceed to payment" button. Saga confirmation worker is explicitly out of scope.

**Tech Stack:** NestJS 11 (ports/DI, ClerkGuard+UserResolver), Drizzle 0.45 + postgres-js, Zod v4 (`@linkz/shared`/`@linkz/config`), SolidJS 1.9 + `@solidjs/router`, Vitest 4, Playwright 1.61 (`@nx/playwright`), Stripe Checkout Sessions, pnpm + Nx 23.

## Global Constraints

- Run all tasks through nx: `pnpm nx run <proj>:<target>`, `pnpm nx run-many -t test|lint|build`.
- TypeScript strict; no `any` (use `unknown` + narrow); no `!` non-null assertions; prefer union literals + `as const`; derive types via `z.infer`; validate external input with Zod at the boundary.
- NestJS: thin controllers, logic in services; depend on injected ports (`Symbol` tokens), not concretions; one module per feature; read config only through `@linkz/config`; surface errors through the global exception filter (throw NestJS `HttpException` subclasses).
- SolidJS: never destructure `props` (read `props.x`); signals + `createMemo`/`createResource`; `<Show>`/`<For>` not ternaries/`.map`; components small and presentational; reach API through a thin client.
- Seat-concurrency invariant: at most one active reservation per seat via the partial-unique index `uq_active_seat_partial` — never replace with app-level checks alone.
- Webhook idempotency: confirm a reservation only on `payment.succeeded`; process each event once via the `webhook_events` ledger.
- Hold TTL = 15 minutes; pricing = `seats.price` (minor units) + `SERVICE_FEE_AMOUNT` (default 400); currency `usd`.
- Integration tests that hit Postgres use a separate `linkz_test` DB (see memory note); never rely on the shared dev DB for assertions.
- TDD: write the failing test, run it red, implement minimal code, run green, commit. Frequent commits with the `Co-Authored-By: Claude <noreply@anthropic.com>` trailer.

```ts
import { ForbiddenException, GoneException, ConflictException } from '@nestjs/common';
import type { ReservationStatus } from '@linkz/shared';

export interface Reservation {
  id: string;
  seatId: string;
  userId: string;
  status: ReservationStatus;
  sagaStep: string;
  holdExpiresAt: Date | null;
  version: number;
}

export function isHoldExpired(res: Reservation, now: Date): boolean {
  return res.holdExpiresAt !== null && res.holdExpiresAt.getTime() <= now.getTime();
}

export function assertOwns(res: Reservation, userId: string): void {
  if (res.userId !== userId) throw new ForbiddenException('Reservation does not belong to caller');
}

export function assertActiveHold(res: Reservation, now: Date): void {
  if (isHoldExpired(res, now)) throw new GoneException('Seat hold has expired');
  if (res.status !== 'held') throw new ConflictException(`Reservation is ${res.status}, not held`);
}
```

```ts
import { Injectable } from '@nestjs/common';
import { eq, and } from 'drizzle-orm';
import { reservations } from '@linkz/database';
import { DrizzleService } from '../../drizzle/drizzle.service';
import type { ReservationsRepositoryPort } from '../reservations.ports';
import type { Reservation } from '../reservation.domain';

@Injectable()
export class DrizzleReservationsRepository implements ReservationsRepositoryPort {
  constructor(private readonly drizzle: DrizzleService) {}

  private get db() {
    return this.drizzle.db;
  }

  async hold(seatId: string, userId: string, holdExpiresAt: Date): Promise<Reservation | null> {
    try {
      const [row] = await this.db
        .insert(reservations)
        .values({
          seatId,
          userId,
          status: 'held',
          sagaStep: 'hold',
          holdExpiresAt,
        })
        .returning();
      return this.toReservation(row);
    } catch (e) {
      // 23505 = postgres unique_violation. The partial-unique index
      // uq_active_seat_partial is the hard concurrency guard.
      if (e && typeof e === 'object' && 'code' in e && (e as { code: string }).code === '23505') {
        return null;
      }
      throw e;
    }
  }

  async findById(id: string): Promise<Reservation | null> {
    const [row] = await this.db.select().from(reservations).where(eq(reservations.id, id));
    return row ? this.toReservation(row) : null;
  }

  async findOwnedBy(userId: string): Promise<Reservation[]> {
    const rows = await this.db.select().from(reservations).where(eq(reservations.userId, userId));
    return rows.map((r) => this.toReservation(r));
  }

  async release(id: string, version: number): Promise<boolean> {
    const [row] = await this.db
      .update(reservations)
      .set({ status: 'released', sagaStep: 'released', updatedAt: new Date() })
      .where(and(eq(reservations.id, id), eq(reservations.version, version), eq(reservations.status, 'held')))
      .returning();
    return !!row;
  }

  private toReservation(row: typeof reservations.$inferSelect): Reservation {
    return {
      id: row.id,
      seatId: row.seatId,
      userId: row.userId,
      status: row.status,
      sagaStep: row.sagaStep,
      holdExpiresAt: row.holdExpiresAt,
      version: row.version,
    };
  }
```

```ts
import { Inject, Injectable, NotFoundException, ConflictException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import {
  CheckoutReviewResponseSchema,
  ReservationResponseSchema,
  type CheckoutReviewResponse,
  type ReservationResponse,
} from '@linkz/shared';
import { RESERVATIONS_REPOSITORY, type ReservationsRepositoryPort } from './reservations.ports';
import { SEATS_REPOSITORY, type SeatsRepository, type SeatWithStatus } from '../seats/seats.ports';
import { assertOwns, assertActiveHold, type Reservation } from './reservation.domain';
import { computePricing } from '../pricing/pricing';

@Injectable()
export class ReservationsService {
  constructor(
    @Inject(RESERVATIONS_REPOSITORY) private readonly repo: ReservationsRepositoryPort,
    @Inject(SEATS_REPOSITORY) private readonly seats: SeatsRepository,
    private readonly config: ConfigService,
  ) {}

  async holdSeat(seatId: string, userId: string): Promise<ReservationResponse> {
    const seat = await this.seats.findById(seatId);
    if (!seat) throw new NotFoundException(`Seat ${seatId} not found`);
    const ttlMinutes = this.config.get<number>('HOLD_TTL_MINUTES') ?? 15;
    const holdExpiresAt = new Date(Date.now() + ttlMinutes * 60_000);
    const res = await this.repo.hold(seatId, userId, holdExpiresAt);
    if (!res) throw new ConflictException('Seat is already held');
    return this.toReservationResponse(res);
  }

  async getCheckoutReview(reservationId: string, userId: string, email: string): Promise<CheckoutReviewResponse> {
    const res = await this.repo.findById(reservationId);
    if (!res) throw new NotFoundException(`Reservation ${reservationId} not found`);
    assertOwns(res, userId);
    assertActiveHold(res, new Date());
    const seat = await this.seats.findById(res.seatId);
    if (!seat) throw new NotFoundException(`Seat ${res.seatId} not found`);
    const fee = this.config.get<number>('SERVICE_FEE_AMOUNT') ?? 400;
    const currency = this.config.get<string>('SERVICE_FEE_CURRENCY') ?? 'usd';
    const pricing = { ...computePricing(seat.price, fee), currency };
    return CheckoutReviewResponseSchema.parse({
      reservation: this.toReservationResponse(res),
      seat: { id: seat.id, label: seat.label, descriptor: seatDescriptor(seat), price: seat.price },
      pricing,
      contact: { email },
    });
  }

  async release(reservationId: string, userId: string): Promise<void> {
    const res = await this.repo.findById(reservationId);
    if (!res) throw new NotFoundException(`Reservation ${reservationId} not found`);
    assertOwns(res, userId);
    const ok = await this.repo.release(res.id, res.version);
    if (!ok) throw new ConflictException(`Reservation is ${res.status}, cannot release`);
  }

  async findOwnedBy(userId: string): Promise<ReservationResponse[]> {
    const rows = await this.repo.findOwnedBy(userId);
    return rows.map((r) => this.toReservationResponse(r));
  }

  private toReservationResponse(res: Reservation): ReservationResponse {
    return ReservationResponseSchema.parse({
      id: res.id,
      seatId: res.seatId,
      status: res.status,
      expiresAt: res.holdExpiresAt ? res.holdExpiresAt.toISOString() : null,
    });
  }
}

function seatDescriptor(seat: SeatWithStatus): string {
  // Front/middle/back row descriptor derived from label like "A4".
  const row = seat.label.match(/^([A-Z])/)?.[1] ?? '';
  const position = row <= 'B' ? 'Front row' : row <= 'D' ? 'Middle row' : 'Back row';
  return `${position}`;
}
```

```ts
import { Inject, Injectable, ConflictException, NotFoundException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { authorizeCreatePayment, mapEvent } from './payments.domain';
import {
  PAYMENT_PROVIDER,
  PAYMENTS_REPOSITORY,
  type PaymentProviderPort,
  type PaymentsRepositoryPort,
  type PaymentRow,
} from './payments.ports';
import { RESERVATIONS_REPOSITORY, type ReservationsRepositoryPort } from '../reservations/reservations.ports';
import { SEATS_REPOSITORY, type SeatsRepository } from '../seats/seats.ports';
import { computePricing } from '../pricing/pricing';
import { PaymentResponseSchema } from '@linkz/shared';

@Injectable()
export class PaymentsService {
  constructor(
    @Inject(PAYMENT_PROVIDER) private readonly provider: PaymentProviderPort,
    @Inject(PAYMENTS_REPOSITORY) private readonly repo: PaymentsRepositoryPort,
    @Inject(RESERVATIONS_REPOSITORY) private readonly reservations: ReservationsRepositoryPort,
    @Inject(SEATS_REPOSITORY) private readonly seats: SeatsRepository,
    private readonly config: ConfigService,
  ) {}

  async createPayment(reservationId: string, userId: string): Promise<{ paymentId: string; checkoutUrl: string }> {
    const reservation = await this.reservations.findById(reservationId);
    if (!reservation) throw new NotFoundException(`Reservation ${reservationId} not found`);
    authorizeCreatePayment(reservation, userId, new Date());

    const existing = await this.repo.findByReservation(reservationId);
    if (existing?.status === 'succeeded') {
      throw new ConflictException('Reservation already paid');
    }

    const seat = await this.seats.findById(reservation.seatId);
    if (!seat) throw new NotFoundException(`Seat ${reservation.seatId} not found`);
    const fee = this.config.get<number>('SERVICE_FEE_AMOUNT') ?? 400;
    const currency = this.config.get<string>('SERVICE_FEE_CURRENCY') ?? 'usd';
    const pricing = computePricing(seat.price, fee);
    const amount = pricing.total;
    const idempotencyKey = `pay_${reservationId}`;

    const payment: PaymentRow =
      existing ??
      (await this.repo.insertPending({
        reservationId,
        provider: 'stripe',
        providerRef: null,
        idempotencyKey,
        amount,
        currency,
        status: 'pending',
      }));

    const { providerRef, checkoutUrl } = await this.provider.createCheckoutSession({
      paymentId: payment.id,
      reservationId,
      amount: payment.amount,
      currency: payment.currency,
      idempotencyKey: payment.idempotencyKey,
    });
    await this.repo.updateProviderRef(payment.id, providerRef);

    return { paymentId: payment.id, checkoutUrl };
  }

  async getPaymentByReservation(reservationId: string, userId: string) {
    const reservation = await this.reservations.findById(reservationId);
    if (!reservation) throw new NotFoundException(`Reservation ${reservationId} not found`);
    if (reservation.userId !== userId) throw new ForbiddenException('Reservation does not belong to caller');
    const payment = await this.repo.findByReservation(reservationId);
    if (!payment) throw new NotFoundException('No payment for reservation');
    return PaymentResponseSchema.parse({
      id: payment.id,
      reservationId: payment.reservationId,
      status: payment.status,
      amount: payment.amount,
      currency: payment.currency,
    });
  }

  async handleWebhook(rawBody: Buffer | string, signature: string): Promise<void> {
    const event = this.provider.constructEvent(rawBody, signature);
    const outcome = mapEvent(event);
    if (outcome === 'ignore' || !event.paymentId) return;
    await this.repo.settle(event.paymentId, outcome, event);
  }
}
```

```ts
import type { ZodType } from 'zod';

const BASE = (import.meta.env.VITE_API_URL as string | undefined) ?? '/api';

export class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message);
  }
}

export interface ApiFetchOpts<T> {
  method?: string;
  body?: unknown;
  schema: ZodType<T>;
  getToken: () => Promise<string | null>;
}

export async function apiFetch<T>(path: string, opts: ApiFetchOpts<T>): Promise<T> {
  const token = await opts.getToken();
  if (!token) throw new Error('Not authenticated');
  const res = await fetch(`${BASE}${path}`, {
    method: opts.method ?? 'GET',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
    body: opts.body !== undefined ? JSON.stringify(opts.body) : undefined,
  });
  if (!res.ok) {
    let message = `request failed: ${res.status}`;
    try { const b = await res.json(); if (b?.message) message = b.message; } catch { /* ignore */ }
    throw new ApiError(res.status, message);
  }
  const raw = await res.json();
  return opts.schema.parse(raw);
}
```
